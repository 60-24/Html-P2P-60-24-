package core

import (
    "crypto/ed25519"
    "time"
)

type Event struct {
    ID        string            `json:"id"`
    Time      int64             `json:"time"`
    Action    string            `json:"action"`
    Entity    string            `json:"entity"`
    Space     string            `json:"space"`
    Intent    string            `json:"intent"`
    Payload   map[string]interface{} `json:"payload"`
    Signature string            `json:"sig"`
    CausalityVector []string    `json:"causality"`
}

type EventBus struct {
    events chan Event
    store  *BadgerStore
    pubKey ed25519.PublicKey
}

func (eb *EventBus) Emit(e Event) error {
    // Sign event
    sig := ed25519.Sign(eb.store.privKey, marshalEvent(e))
    e.Signature = base64.StdEncoding.EncodeToString(sig)
    
    // Store + broadcast
    eb.store.Put(e.ID, e)
    eb.events <- e
    return nil
}
type IntentResolver struct {
    validators map[string]ValidatorFunc
}

func (ir *IntentResolver) Resolve(e Event) (Resolution, error) {
    validator := ir.validators[e.Intent]
    if validator == nil {
        return Resolution{}, ErrUnknownIntent
    }
    
    // 1. Validate signature
    if !VerifySignature(e) {
        return Resolution{}, ErrBadSignature
    }
    
    // 2. Check causality
    if !ir.checkCausality(e) {
        return Resolution{}, ErrBadOrdering
    }
    
    // 3. Run intent logic
    ok, err := validator(e)
    return Resolution{OK: ok, Err: err}, err
}

type Resolution struct {
    OK   bool
    Err  error
    Next map[string]interface{}
}
type GossipNode struct {
    id       string
    peers    []string  // peer addresses
    port     int       // 6024
    conn     *net.UDPConn
    eventBus *EventBus
}

func (gn *GossipNode) BroadcastEvent(e Event) error {
    data, _ := json.Marshal(e)
    
    for _, peer := range gn.peers {
        addr, _ := net.ResolveUDPAddr("udp", peer)
        gn.conn.WriteToUDP(data, addr)
    }
    return nil
}

func (gn *GossipNode) ListenForEvents() {
    buf := make([]byte, 4096)
    
    for {
        n, remoteAddr, _ := gn.conn.ReadFromUDP(buf)
        var e Event
        json.Unmarshal(buf[:n], &e)
        
        // Verify & merge
        gn.eventBus.Emit(e)
        
        // Forward to other peers (flood with dedup)
        gn.BroadcastEvent(e)
    }
}
type StateEngine struct {
    db     *badger.DB
    states map[string]State
    clock  *LamportClock
}

type State struct {
    ID       string
    Data     map[string]interface{}
    Version  int64
    Checksum string
}

func (se *StateEngine) ApplyEvent(e Event) error {
    // 1. Increment clock
    se.clock.Increment()
    
    // 2. Create state snapshot
    prevState := se.states[e.Entity]
    newState := State{
        ID:      fmt.Sprintf("s_%s_%d", e.Entity, se.clock.Value()),
        Data:    applyAction(prevState.Data, e.Action, e.Payload),
        Version: se.clock.Value(),
    }
    
    // 3. Persist to BadgerDB
    key := []byte(newState.ID)
    val, _ := json.Marshal(newState)
    se.db.Set(key, val)
    
    // 4. Update in-memory map
    se.states[e.Entity] = newState
    
    return nil
}

7️⃣ Consensus (Byzantine Fault Tolerant)

1. Propose
Node A publishes Event → signed, timestamped, causally linked
2. Gossip
Flood broadcast UDP 6024 → all reachable nodes
3. Validate
Each node: verify signature, check causality, validate intent
4. Merge
CRDT merge → state reconciliation (LWW / vector clock)
5. Ack
Send ACK gossip → propagate acceptance (implicit consensus)
6. Finalize
Once majority ACKs → state committed to BadgerDB

⚠️ Kluczowe decyzje architektoniczne

Brak liderów
Każdy node jest równouprawnionym peer'em — żaden RAFT/leader election
Implicit Consensus
Konsensus to efekt, nie proces — wynika z verify + flood
Causality First
Lamport Clock + vector clocks → gwarantowana konsystencja zdarzeń
State = Projection
State to zawsze projekcja historii zdarzeń — single source of truth = Event Log
Intent-Driven
Nie "Execute Command" → "Why Execute This" (intent) — każda akcja ma cel
